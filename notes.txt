create postgres db
docker run --name news-postgres \
  -e POSTGRES_USER=postgres \
  -e POSTGRES_PASSWORD=password \
  -e POSTGRES_DB=news_dashboard \
  -p 5432:5432 \
  -d postgres:16-alpine

set connection string in .env

DATABASE_URL="postgresql://postgres:password@localhost:5432/news_dashboard?schema=public"
